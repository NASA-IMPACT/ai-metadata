% ---------------------------------------------------------------- pitch
\begin{frame}{The pitch we made}
  \begin{quote}
    \large The best representation is \textit{\textcolor{good}{not universal}} ---
    it is consumption-mode- and model-dependent, and must be measured empirically.
  \end{quote}
  \medskip
  \muted{Made falsifiable by crossing \textbf{format} against \textbf{content scope} ---
  the confound the field has been ignoring.}
  \bigskip
  \begin{columns}[t]
    \column{0.5\textwidth}
      \textbf{\textcolor{good}{H1}}\; \textbf{Content dominates}\\
      \muted{\small field selection matters more than format for retrieval.}
      \medskip\par
      \textbf{\textcolor{good}{H3}}\; \textbf{The axes disagree}\\
      \muted{\small retrieval, reasoning, and cost need not rank formats alike.}
    \column{0.5\textwidth}
      \textbf{\textcolor{good}{H2}}\; \textbf{Format effect is small}\\
      \muted{\small at equal content, representations barely differ.}
      \medskip\par
      \textbf{\textcolor{good}{H4}}\; \textbf{Best format varies}\\
      \muted{\small the top reasoning representation differs by model family.}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------- setup
\begin{frame}{Experimental setup --- the grid}
  \begin{columns}[t]
    \column{0.56\textwidth}
    \textbf{Three independent axes}
    \begin{itemize}
      \item \textbf{Payload} --- \textcolor{accent}{faceted}: curated 32-field projection
        \;vs.\; \textcolor{accent2}{unfaceted}: raw UMM record (45 top-level fields, 398 paths)
      \item \textbf{Format} --- JSON, YAML, TOON, CSV, JSON-LD, Metadata-as-Text (prose);
        all six carry identical facts (parity hard-gated at render)
      \item \textbf{Model} --- \texttt{gpt-5.4-nano}, \texttt{gpt-5.4-mini} (cloud);
        \texttt{muse-glimmer:30b} (local 32B)
    \end{itemize}
    \medskip
    \textbf{Held fixed} \muted{--- encoder (bge-large-en-v1.5), 510-token chunks,
    max pooling, $k{=}10$ records in context, temperature 0, seed, one judge}
    \column{0.42\textwidth}
    \textbf{Scale}
    \begin{itemize}
      \item 2{,}584 real CMR collection records $\times$ 12 renderings
      \item 511 queries for retrieval \muted{(11 SME + 500 synthetic)}
      \item 300-query slice for LLM stages
      \item $6 \times 2 \times 3 = 36$ arms\\ $\times$ 300 queries = \textbf{10{,}800 judged cells}
      \item Spend: \$71 OpenAI + ${\sim}145$\,h local generation
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Experimental setup --- staged pipeline, every number auditable}
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{@{}llll@{}}
    \toprule
    Stage & What it does & Measures & LLM involved \\
    \midrule
    \textbf{R} \;Retrieve & chunked vector search per format/payload & Recall@k, MRR, nDCG@10 & none --- exact \\
    \textbf{A} \;Answer & top-10 records into each model, temp 0 & prompt/completion tokens, latency, \$ & the 3 under test \\
    \textbf{J} \;Judge & DeepEval metrics on frozen answers & correctness, faithfulness, \ldots & \texttt{gpt-5.4-nano} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Stages are frozen artifacts --- judging re-runs never touch answers; checkpointed and resumable
    \item Correctness is the end-to-end metric (knows ground truth); the rest are diagnostics
    \item Content parity across formats is enforced, so format effects are representation-only
  \end{itemize}
  \bigskip
  \centering\muted{The same 60\% token gap shows up before the LLM ever runs: faceted renders at
  786--1{,}465 bge tokens/record; unfaceted at 1{,}462--7{,}012.}
\end{frame}

% ---------------------------------------------------------------- H1
\begin{frame}{H1 --- Content dominates \hfill \verdict{Supported}}
  \textbf{The payload axis is the largest effect in the study, on all three measurement axes.}
  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    Axis & Faceted vs.\ unfaceted & Largest format effect & Ratio \\
    \midrule
    Reasoning (correctness) & $+0.045$ $[+0.038,+0.052]$ & CSV $-0.027$ vs.\ JSON & ${\sim}1.7\times$ \\
    Retrieval (R@10) & $+0.01$ to $+0.14$, every format & JSON-LD $-0.07$ faceted & similar$^{\dagger}$ \\
    Cost (prompt tokens) & $-16{,}533$/query \;(${-}60\%$) & ${\sim}{-}3{,}500$ within payload & ${\sim}5\times$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Per model: nano $+0.049$, mini $+0.051$, glimmer $+0.036$ --- all clear of zero at $n{=}1{,}800$
    \item The curated projection is \textbf{better and cheaper for every model tested} --- the
      run's headline finding, and it is not close
    \item $^{\dagger}$Retrieval caveat: the synthetic query generator saw a field summary that
      structurally resembles the faceted projection --- payload gap there is an observation, not a finding
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- H2
\begin{frame}{H2 --- Format effect is small \hfill \verdict{Supported}\muted{, with one outlier}}
  \textbf{At equal content, serialization moves correctness by at most half the payload effect.}
  \bigskip
  \begin{columns}[t]
    \column{0.52\textwidth}
    \textbf{Reasoning, paired vs.\ JSON} \muted{($n{=}1{,}800$ each)}
    \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{@{}lr@{}}
      \toprule
      MaT (prose) & $+0.009$ \;\muted{$[-0.003,+0.020]$} \\
      TOON & $-0.008$ \;\muted{$[-0.019,+0.003]$} \\
      YAML & $-0.011$ \;\muted{$[-0.022,-0.000]$} \\
      JSON-LD & $-0.015$ \;\muted{$[-0.027,-0.003]$} \\
      CSV & $-0.027$ \;\muted{$[-0.039,-0.015]$} \\
      \bottomrule
    \end{tabular}
    \end{center}
    \column{0.46\textwidth}
    \begin{itemize}
      \item Only CSV and JSON-LD clearly separate from JSON --- and by less than
        the payload's $0.045$
      \item \textbf{Format matters far less than what you put in it}
      \item The outlier: at \textit{retrieval}, JSON-LD trails by $0.07$--$0.10$ R@10
        while costing the most tokens --- format is not \textit{free} at the embedding stage
    \end{itemize}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------- H3
\begin{frame}{H3 --- The axes disagree \hfill \verdict{Supported}}
  \textbf{No single format wins retrieval, reasoning, and cost at once --- rankings genuinely diverge.}
  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \small
  \begin{tabular}{@{}lllll@{}}
    \toprule
    Format & Retrieval (chunked) & Reasoning, faceted & Reasoning, raw & Cost \\
    \midrule
    MaT (prose) & \textbf{1st, all 4 conditions} & top-two, all models & mid & \textbf{cheapest} \\
    JSON & 5th of 6 (faceted) & mid & \textbf{1st, all 3 models} & mid \\
    CSV & 4th faceted / \textbf{last} raw & worst pooled penalty & last (2 of 3) & high (raw) \\
    JSON-LD & \textbf{last} everywhere & mid-pack & mid & \textbf{highest} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Even the judge's diagnostics disagree: contextual recall runs \textit{counter} to correctness
      (faceted MaT 0.36 vs.\ raw JSON-LD 0.60 --- while beating it 0.847 to 0.796 on correctness)
    \item Consequence: a repository cannot pick one representation off a single benchmark axis ---
      exactly the claim in the pitch
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- H4
\begin{frame}{H4 --- Best format varies by model \hfill \verdictwarn{Suggestive}}
  \textbf{Three models, three different winners under the curated payload --- but the gaps are
  inside the error bars.}
  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{@{}lll@{}}
    \toprule
    Model & Faceted winner & Raw-record winner \\
    \midrule
    \texttt{gpt-5.4-nano} & YAML \,(0.850) & JSON \,(0.811) \\
    \texttt{gpt-5.4-mini} & MaT \,(0.782) & JSON \,(0.740) \\
    \texttt{muse-glimmer:30b} & TOON \,(0.864) & JSON \,(0.837) \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Under the \textbf{raw record, JSON wins for all three} --- explicit structure appears to
      help when there are 398 paths to navigate; content scope \textit{modulates} format sensitivity
    \item Within-model spreads (0.028--0.040) overlap the paired CIs ($\pm$0.011--0.012):
      the \textit{ordering} is suggestive, not established --- H4 needs more SME-hard queries to close
    \item What \textit{is} established: no format is safe to hard-code across models
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- model axis
\begin{frame}{The model axis --- interesting, and least trustworthy}
  \begin{columns}[t]
    \column{0.46\textwidth}
    \textbf{Paired correctness} \muted{($n{=}3{,}600$ each)}
    \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{@{}lr@{}}
      \toprule
      mini $-$ nano & $-0.076$ \\
      glimmer $-$ nano & $+0.022$ \\
      mini $-$ glimmer & $-0.098$ \\
      \bottomrule
    \end{tabular}
    \end{center}
    \medskip
    \muted{\small Read literally: a local 32B beats both cloud models, and the small cloud model
    beats the larger one.}
    \column{0.52\textwidth}
    \textbf{Before believing that:}
    \begin{itemize}
      \item The judge \textbf{is} \texttt{gpt-5.4-nano} --- it grades its own answers in two of
        three comparisons, and wins both. Self-preference cannot be separated from quality here
      \item The one \textbf{judge-neutral} comparison (mini vs.\ glimmer) is the \textit{largest}
        effect --- and glimmer's win over nano is biased \textit{against} glimmer, so it is conservative
      \item Slice size matters: glimmer's edge read $+0.067$ at 50 queries, $+0.022$ at 300 ---
        the small slice was SME-heavy, glimmer's best terrain
    \end{itemize}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------- SME
\begin{frame}{The hard part: expert questions}
  \textbf{SME-written queries score $\sim$0.3--0.4 correctness below synthetic ones, for every model.}
  \bigskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Model \muted{(faceted, range over formats)} & SME ($n{=}11$) & Synthetic ($n{=}289$) & Gap \\
    \midrule
    \texttt{gpt-5.4-nano} & 0.36--0.48 & 0.84--0.87 & ${\sim}0.43$ \\
    \texttt{gpt-5.4-mini} & 0.28--0.42 & 0.76--0.80 & ${\sim}0.40$ \\
    \texttt{muse-glimmer:30b} & \textbf{0.47--0.63} & 0.85--0.87 & ${\sim}0.31$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Synthetic queries inherit vocabulary from the records they were generated from;
      expert questions do not --- \textbf{the SME slice is the realistic difficulty}
    \item Pooled means are dominated by slice composition; compare per-slice, never pooled
    \item The local model has the smallest expert gap and the best expert scores --- worth chasing,
      but $n{=}11$ is far too few expert queries to conclude anything
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- caveats
\begin{frame}{What we would push back on ourselves}
  \begin{itemize}
    \item \textbf{The judge is a model under test.} All quality scores are nano-graded;
      the model ranking is the result we trust least. \muted{(The format and payload effects are
      within-judge comparisons and far more robust.)}
    \item \textbf{Query provenance leans faceted, and ${\sim}3/4$ of the payload effect is
      retrieval-mediated.} The synthetic generator consumed a curated field summary, and each
      payload retrieves its own contexts. \muted{With retrieval equalized (both payloads at
      R@10 $=$ 1), faceted still wins --- but by $+0.012$, not $+0.045$.}
    \item \textbf{One encoder, one chunking scheme.} Prose ranked \textit{last} under a different
      encoder on whole documents and first here --- retrieval rankings are
      encoder-conditional. \muted{Chunk boundaries also cut JSON mid-object but prose mid-paragraph.}
    \item \textbf{Format rankings within a model are not statistically separated} --- only the
      pooled CSV and JSON-LD penalties clear their intervals.
    \item \textbf{Expert coverage is thin}: 11 SME queries carry all realistic difficulty.
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- future work
\begin{frame}{Future work}
  \textbf{Close the loops this run opened}
  \begin{itemize}
    \item \textbf{Neutral judge re-run} --- re-grade a slice with mini or glimmer as judge to bound
      self-preference; the pipeline already supports coexisting judges per cell
    \item \textbf{Second encoder} --- one more index (same chunks, different encoder) separates the
      encoder/chunking confound behind prose's retrieval win
    \item \textbf{More SME queries} --- the realistic slice is 11 questions; this is the single
      highest-leverage data investment
  \end{itemize}
  \medskip
  \textbf{Next experiments from the original program}
  \begin{itemize}
    \item \textbf{Field-description ablation} --- bare names vs.\ inline descriptions vs.\ enums;
      the literature predicts the largest single gain (Gorilla, ToolAlpaca: $+30\%$)
    \item \textbf{Direct comprehension} --- deterministic Q\&A over single records, no retrieval, no judge
    \item \textbf{Sparse-metadata stress test} --- ablate fields, measure hallucination vs.\
      honest gap-reporting per representation
    \item \textbf{Agentic/MCP arm} --- static representation vs.\ tool-calling agent
      (\texttt{nasa/earthdata-mcp} already exists)
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------- takeaways
\begin{frame}{Takeaways for CMR}
  \begin{enumerate}
    \setlength{\itemsep}{0.7em}
    \item \textbf{Publish a curated projection as a first-class AI representation.}
      A 32-field facet of UMM was more correct \textit{and} 60\% cheaper than the raw record
      for every model tested. This is the actionable finding, and it is not close.
    \item \textbf{Ship a Metadata-as-Text summary alongside it.}
      Prose won retrieval in every condition, is the cheapest rendering to embed, and stayed
      top-two on reasoning --- the strongest single default for embedding pipelines.
    \item \textbf{Keep the canonical UMM record untouched.}
      The projection is an \textit{added} representation; precision, provenance, and audit
      stay with the record of authority.
    \item \textbf{Do not assume JSON-LD helps LLM consumption.}
      It was the most expensive rendering and the worst retriever here. Its value is
      interoperability and linking --- serve it for that, not as LLM context.
    \item \textbf{Content curation first, format second.}
      Deciding \textit{which fields} go into context beats any serialization debate by a wide margin.
    \item \textbf{Fund the harness, not a format.}
      Winners are model- and encoder-dependent and models change; a repeatable, audited
      evaluation pipeline is the durable artifact. \muted{(This one is open and resumable.)}
  \end{enumerate}
\end{frame}
